Step 1 — Find your conda Python path

conda activate orchestra_env
which python
# Copy the output — it'll look like:
# /home/faizy/miniconda3/envs/orchestra_env/bin/python
# or /home/faizy/anaconda3/envs/orchestra_env/bin/python


Step 2 — Add faizy to every hardware group needed

sudo usermod -aG dialout,video,gpio,spi,i2c,audio faizy

Step 3 — Create the service file
Replace /home/faizy/miniconda3 with your actual conda base path from Step 1:

sudo nano /etc/systemd/system/orchestra.service

Paste this (edit the ExecStart Python path to match your Step 1 output):

`[Unit]
Description=Biometric Orchestrator – Face + Voice
Documentation=https://github.com/yourrepo
After=network.target
Wants=network.target

[Service]
Type=simple
User=faizy
Group=faizy

WorkingDirectory=/home/faizy/Documents/FYP_CodeBase/Orchestra/

# Use the conda env Python directly — no 'conda activate' needed
ExecStart=/home/faizy/miniconda3/envs/orchestra_env/bin/python Orchestra.py

# Clean restart if it crashes (5 s delay, max 5 restarts in 2 min)
Restart=on-failure
RestartSec=5
StartLimitIntervalSec=120
StartLimitBurst=5

# Give hardware time to be ready (camera/UART init)
TimeoutStartSec=30

# Stdout + stderr both go to journald — readable with: journalctl -u orchestra -f
StandardOutput=journal
StandardError=journal

# Hardware env vars that libcamera/picamera2 need in a non-login shell
Environment=PYTHONUNBUFFERED=1
Environment=LIBCAMERA_LOG_LEVELS=*:ERROR
Environment=HOME=/home/faizy

[Install]
WantedBy=multi-user.target`


Step 4 — Enable and start it	

sudo systemctl daemon-reload
sudo systemctl enable orchestra.service    # auto-start at boot
sudo systemctl start orchestra.service     # start it right now without rebooting
