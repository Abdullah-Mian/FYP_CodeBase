cd ~

# 2. Check if ncnn directory exists (from your first clone attempt)
ls -la | grep ncnn

# 3. If it exists, remove it and start fresh
rm -rf ncnn

# 4. Clone ncnn WITHOUT --depth 1 (to avoid submodule issues)
git clone https://github.com/Tencent/ncnn.git

# 5. Enter ncnn directory
cd ncnn

# 6. Initialize submodules
git submodule update --init --recursive


# Install ONNX Python library (contains the protobuf definitions ncnn needs)
sudo apt install -y python3-onnx

# OR install via pip (works better)
pip install onnx==1.15.0 --no-cache-dir




##########################################
# Install the Protobuf Compiler
sudo apt update
sudo apt install -y protobuf-compiler libprotobuf-dev

protoc --version
# Output should be something like: libprotoc 3.21.12
##########################################

# 7. Create build directory
mkdir -p build
cd build

###############
# or if you are going to rebuild ncnn
cd ~/ncnn/build
rm -rf *
###############

cmake -DCMAKE_BUILD_TYPE=Release \
      -DNCNN_VULKAN=ON \
      -DNCNN_BUILD_TOOLS=ON \
      -DNCNN_BUILD_EXAMPLES=OFF \
      -DNCNN_BUILD_BENCHMARK=OFF \
      -DNCNN_BUILD_TESTS=OFF \
      -DNCNN_SYSTEM_GLSLANG=OFF \
      -DNCNN_PYTHON=ON \
      ..

# Look for this line - should now say YES:
# -- Building with onnx = YES

make -j4

# and if not asked Install manually
sudo make install
sudo ldconfig

# Verify ONNX tools are now built
ls ~/ncnn/build/tools/onnx/
###############################################
#proceed with conversion:
cd ~/Downloads/FYP/Mobilefacenet

~/ncnn/build/tools/onnx/onnx2ncnn \
  models/mobilefacenet_sim.onnx \
  models/mobilefacenet.param \
  models/mobilefacenet.bin

~/ncnn/build/tools/ncnnoptimize \
  models/mobilefacenet.param \
  models/mobilefacenet.bin \
  models/mobilefacenet_opt.param \
  models/mobilefacenet_opt.bin \
  0
###############################################
# if onnx not found
# Step 1: Verify ncnn was built
ls ~/ncnn/build/install/lib/libncnn.a
ls ~/ncnn/build/install/include/ncnn/

# Step 2: Manually install to system directories
sudo mkdir -p /usr/local/lib/ncnn
sudo mkdir -p /usr/local/include/ncnn

# Step 3: Copy library
sudo cp ~/ncnn/build/install/lib/libncnn.a /usr/local/lib/ncnn/libncnn.a

# Step 4: Copy headers
sudo cp -r ~/ncnn/build/install/include/ncnn/* /usr/local/include/ncnn/

# Step 5: Copy glslang libraries (needed for Vulkan)
sudo cp ~/ncnn/build/install/lib/libglslang.a /usr/local/lib/
sudo cp ~/ncnn/build/install/lib/libSPIRV.a /usr/local/lib/
sudo cp ~/ncnn/build/install/lib/libMachineIndependent.a /usr/local/lib/
sudo cp ~/ncnn/build/install/lib/libGenericCodeGen.a /usr/local/lib/
sudo cp ~/ncnn/build/install/lib/libOSDependent.a /usr/local/lib/
sudo cp ~/ncnn/build/install/lib/libglslang-default-resource-limits.a /usr/local/lib/

# Step 6: Update library cache
sudo ldconfig

# Step 7: Verify installation
ls /usr/local/lib/ncnn/libncnn.a
ls /usr/local/include/ncnn/net.h
