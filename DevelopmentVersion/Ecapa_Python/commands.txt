Step 1 — Enable hardware UART (do this first):
sudo raspi-config
# Interface Options → Serial Port
#   Login shell over serial: NO
#   Serial port hardware:    YES
# Finish → Reboot

Step 2 — Check your OS bitness:
uname -m   # aarch64 = 64-bit,  armv7l = 32-bit

Step 3a — 64-bit RPi OS (aarch64) — simple:
sudo apt update && sudo apt install -y python3-pip python3-numpy python3-scipy libsndfile1
pip3 install onnxruntime pyserial psutil

Step 3b — 32-bit RPi OS (armhf) — onnxruntime needs a pre-built wheel:
sudo apt install -y python3-pip python3-numpy python3-scipy python3-serial python3-psutil
# Get Python version first: python3 --version  (e.g. 3.11)
pip3 install https://github.com/nknytk/built-onnxruntime-for-raspberrypi-linux/raw/master/wheels/bullseye/onnxruntime-1.16.3-cp311-cp311-linux_armv7l.whl
# Change cp311 to match your Python version (cp39, cp310, cp311, cp312)

Step 4 — Run:
# Copy both files to same directory:
#   rpi4_ecapa_voice_biometric.py
#   ecapa_tdnn.onnx

python3 rpi4_ecapa_voice_biometric.py
# With debug WAV saving:
python3 rpi4_ecapa_voice_biometric.py --debug
